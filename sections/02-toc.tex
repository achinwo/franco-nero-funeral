% Contents. A hairline-ruled list: a roman numeral in the margin, the section
% title, and a one-line gloss beneath it. Kept quiet on purpose -- this page
% is a signpost, not a statement.

\newlength{\tocnumwidth}\setlength{\tocnumwidth}{9mm}
\newlength{\tocpagewidth}\setlength{\tocpagewidth}{8mm}

\newcommand{\tocrule}{%
  \noindent{\color{rulecolor}\rule{\linewidth}{0.2pt}}\par
}

% #1 numeral, #2 title, #3 gloss, #4 label to take the folio from -- and to
% link to: the whole entry, numeral to folio, is one target, so anywhere in
% the row can be clicked.
%
% The row is boxed first and the box is what gets linked. \hyperref will not
% take three minipages as its argument -- it processes the argument in a way
% that loses track of which environment is open, and the \end{minipage} comes
% back as "\begin{document} ended by \end{minipage}". \pageref* rather than
% \pageref, too, because hyperref makes a plain \pageref a link of its own,
% and a link inside a link is undefined.
\newsavebox{\tocrow}
\newcommand{\tocentry}[4]{%
  \sbox{\tocrow}{%
  \begin{minipage}[t]{\tocnumwidth}\raggedright
    \strut{\footnotesize\scshape\color{rulecolor}#1}
  \end{minipage}%
  \begin{minipage}[t]{\dimexpr\linewidth-\tocnumwidth-\tocpagewidth\relax}\raggedright
    \strut{\color{primary}#2}\par
    \vspace{1.2mm}
    {\footnotesize\itshape\color{accent}#3}
  \end{minipage}%
  \begin{minipage}[t]{\tocpagewidth}\raggedleft
    \strut{\color{accent}\pageref*{#4}}
  \end{minipage}%
  }%
  \noindent\hyperref[#4]{\usebox{\tocrow}}\par
}

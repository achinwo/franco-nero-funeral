\setsectiontitle{At the Graveside}

\pageghost{ghost-studio-lo}

\begin{center}
  {\Large\scshape\color{primary} At the Graveside}
  \vspace{2pt}

  {\small\color{accent}\textit{Friday 4\textsuperscript{th} September, 2026 --- following the Funeral Mass}}
  \thinseparator
\end{center}

\vspace{1mm}

\begin{multicols}{2}
  \small\raggedright
  \setlength{\parindent}{0pt}
  \raggedcolumns

  \liturgyhead{Blessing of the Grave}

  \priest{Lord God, we give you thanks and praise, for you are God our
    Saviour, who have set the stars in their places. When mankind was caught
    in the snare of death, you set us free through baptism; and in fulfilment
    of your will, our Lord Jesus Christ conquered death and rose to life, to
    bring salvation and resurrection to those who belong to Him by faith.}

  \priest{We ask you, Lord, to bless this grave; give our brother, Francis
    Atsekhameh, peace and rest, and on the day of judgment raise him up to
    eternal life with all the saints. We ask this through Christ our Lord.}
  \allsay{Amen.}

  \rubric{The Priest sprinkles the grave with holy water.}

  \liturgyhead{The Committal}

  \priest{It has pleased Almighty God to call our brother, Francis
    Atsekhameh, from this life to Himself. Accordingly, we commit his body to
    the earth from which it came, since Christ, the first fruits of the dead,
    has risen again and will refashion our frail bodies. We commend our
    brother to the Lord. May He embrace him in His peace and bring his body to
    life again on the last day.}
  \allsay{Amen.}

  \rubric{The body is lowered into the grave.}

  \priest{Lord, have mercy.}
  \allsay{Lord, have mercy.}

  \priest{Christ, have mercy.}
  \allsay{Christ, have mercy.}

  \priest{Lord, have mercy.}
  \allsay{Lord, have mercy.}

  \rubric{After a brief silence, the Our Father is said.}

  \liturgyhead{Intercessions}

  \priest{Let us pray for our brother, Francis Atsekhameh. Let us pray to our
    Lord Jesus Christ, who said: ``I am the resurrection and the life. He who
    believes in me, though he dies, yet shall he live; and whoever believes in
    me shall never die.''}

  \priest{Lord, you consoled Martha and Mary in their distress; draw near to
    us who mourn for Francis Atsekhameh, and dry the tears of those who weep.
    We pray to the Lord.}
  \allsay{Lord, hear our prayer.}

  \priest{Lord, you wept at the death of Lazarus, your friend; comfort us in
    our sorrow. We pray to the Lord.}
  \allsay{Lord, hear our prayer.}

  \priest{Remember his good works done in faith. We pray to the Lord.}
  \allsay{Lord, hear our prayer.}

  \priest{You raised the dead to life; give your son, Francis Atsekhameh,
    eternal life. We pray to the Lord.}
  \allsay{Lord, hear our prayer.}

  \priest{You promised paradise to the thief who repented; bring your son,
    Francis Atsekhameh, to the joys of heaven. We pray to the Lord.}
  \allsay{Lord, hear our prayer.}

  \priest{Francis Atsekhameh was washed clean in baptism and anointed with the
    oil of salvation; give him fellowship with all the saints. We pray to the
    Lord.}
  \allsay{Lord, hear our prayer.}

  \priest{Francis Atsekhameh was nourished with Christ's Body and Blood; grant
    him a place at the table in your heavenly kingdom. We pray to the Lord.}
  \allsay{Lord, hear our prayer.}

  \liturgyhead{The Lord's Prayer}

  \allsay{Our Father, who art in heaven\ldots}

  \liturgyhead{Concluding Prayer}

  \priest{Let us pray. Show your mercy to your departed servant, Francis
    Atsekhameh. Since he strove to do your will, let him not be punished for
    his wrongdoing; and as he was united in true faith with all your believing
    people, in your loving goodness let him be united with your angels and
    saints. Through Christ our Lord.}
  \allsay{Amen.}

\end{multicols}

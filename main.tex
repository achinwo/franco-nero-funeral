% --- Liturgy typesetting ---------------------------------------------------
% Shared by the vigil and the funeral mass. Two rules govern the look: the
% congregation's own words are the only bold text on the page, so anyone can
% find their part at a glance, and rubrics (stage directions) are italic and
% grey so they read as instructions rather than as something to say aloud.

% A major division of the rite (Liturgy of the Word, Liturgy of the
% Eucharist): centred, letterspaced, fenced above and below.
\makeatletter
\newcommand{\liturgypart}[1]{%
  \par\addvspace{5mm}%
  \penalty-300 % a column may break here in preference to below the head
  \noindent\begin{minipage}{\linewidth}
    {\color{rulecolor}\rule{\linewidth}{0.3pt}}\par
    \vspace{1.6mm}
    \centering{\footnotesize\scshape\color{primary}\so{#1}}\par
    \vspace{1.6mm}
    {\color{rulecolor}\rule{\linewidth}{0.3pt}}
  \end{minipage}%
  \par\nobreak\vspace{2.5mm}%
  \@afterheading
}

% A movement of the liturgy: letterspaced caps over a hairline. \@afterheading
% keeps the head welded to the text beneath it, so a column never breaks
% between the two and strands a title at the foot.
\newcommand{\liturgyhead}[1]{%
  \par\addvspace{4mm}%
  \penalty-300
  \noindent\begin{minipage}{\linewidth}
    \raggedright
    {\footnotesize\scshape\color{primary}\so{#1}}\par
    \vspace{1mm}
    {\color{rulecolor}\rule{\linewidth}{0.3pt}}
  \end{minipage}%
  \par\nobreak\vspace{2mm}%
  \@afterheading
}
\makeatother

% A spoken line. The speaker sits in the hanging indent so that wrapped text
% aligns under itself rather than under the label.
% The body is grouped: #2 sets \bfseries for the congregation's parts, and
% without the group that weight leaks past \par into whatever follows.
\newcommand{\liturgyline}[2]{%
  \par
  {%
    \noindent\hangindent=1.1em\hangafter=1
    {\scshape\color{accent}#1}\hskip 0.5em\relax #2\par
  }%
  \vspace{1.3mm}%
}
\newcommand{\priest}[1]{\liturgyline{Priest}{\color{primary}#1}}
\newcommand{\allsay}[1]{\liturgyline{All}{\bfseries\color{primary}#1}}
\newcommand{\reader}[2]{\liturgyline{#1}{\color{primary}#2}}
% A versicle with no speaker label -- set flush with the other spoken text.
\newcommand{\versicle}[1]{%
  \par
  {\noindent\hangindent=1.1em\hangafter=1 \color{primary}#1\par}%
  \vspace{1.3mm}%
}

% A rubric: an instruction to the assembly, never spoken.
\newcommand{\rubric}[1]{%
  \par\vspace{0.6mm}%
  \noindent{\footnotesize\itshape\color{accent}#1}\par
  \vspace{1.6mm}%
}

% Hymn title, then verses. \hymnverse takes a marker (a number, or empty for
% the chorus) and the lines, separated by \\.
\newcommand{\hymntitle}[1]{%
  \par\addvspace{1mm}%
  \noindent{\itshape\color{primary}#1}\par
  \vspace{2mm}%
}
% Inside a verse each \\ starts a fresh paragraph, so that \hangindent
% applies to every line: a line too long for the column then wraps into a
% hanging indent, the way a hymnal sets it, instead of orphaning a word
% hard against the left margin.
\newcommand{\versebody}{%
  \raggedright
  \setlength{\parindent}{0pt}\setlength{\parskip}{0pt}%
  \everypar{\hangindent=1em\hangafter=1}%
  \let\\\par
}
\newcommand{\hymnverse}[2]{%
  \par\addvspace{1.8mm}%
  \noindent
  \makebox[1.2em][l]{\footnotesize\color{accent}#1}%
  \begin{minipage}[t]{\dimexpr\linewidth-1.2em\relax}
    \versebody\color{primary}#2\par
  \end{minipage}\par
}
\newcommand{\hymnchorus}[1]{%
  \par\addvspace{1.8mm}%
  \noindent
  \makebox[1.2em][l]{}%
  \begin{minipage}[t]{\dimexpr\linewidth-1.2em\relax}
    \versebody\itshape\color{accent}#1\par
  \end{minipage}\par
}

\newcommand{\psalmcue}{\hspace{0.4em}{\normalfont\footnotesize\itshape\color{accent}R.}}

\newcommand{\kyrieline}[2]{%

\newcommand{\respond}[1]{\liturgyline{R.}{\bfseries\color{primary}#1}}

% A sung or recited congregational text set in lines rather than as prose

\newcommand{\allverse}[1]{%

% --- Photographs -----------------------------------------------------------
% The four surviving prints are old, damaged and unevenly toned; they are
% cropped and toned to one warm sepia by assets/make-plates.sh, which also
% builds everything below. Re-run that script if the originals change.

\definecolor{cornerink}{HTML}{6B5B4B}  % the dark card of an album's photo corners

% A print mounted on the page: a hairline frame, four photo corners, and an
% italic caption beneath. The corners are what make it read as an album page
% rather than as a figure in a report; the frame alone is not enough, because
% these scans have pale edges that dissolve into the paper without it.
\newlength{\mountwidth}
\newlength{\cornerlen}
\newcommand{\mountedphoto}[3][\linewidth]{%
  \setlength{\mountwidth}{#1}%
  \setlength{\cornerlen}{0.055\mountwidth}%
  \begin{minipage}{\mountwidth}
    \centering
    \begin{tikzpicture}[inner sep=0pt,outer sep=0pt]
      \node (P) {\includegraphics[width=\mountwidth]{#2}};
      \draw[rulecolor,line width=0.3pt]
        (P.south west) rectangle (P.north east);
      \fill[cornerink]
        (P.south west) -- ++(\cornerlen,0) -- ++(-\cornerlen,\cornerlen) -- cycle;
      \fill[cornerink]
        (P.south east) -- ++(-\cornerlen,0) -- ++(\cornerlen,\cornerlen) -- cycle;
      \fill[cornerink]
        (P.north east) -- ++(-\cornerlen,0) -- ++(\cornerlen,-\cornerlen) -- cycle;
      \fill[cornerink]
        (P.north west) -- ++(\cornerlen,0) -- ++(-\cornerlen,-\cornerlen) -- cycle;
    \end{tikzpicture}\par
    \vspace{2.2mm}
    {\footnotesize\itshape\color{accent}#3\par}
  \end{minipage}%
}

% --- The family album ------------------------------------------------------
% assets/build-album.py reads assets/images/family_pics/, solves a
% justified-rows layout across as many pages as it needs, and writes
% plates/family-album.tex as a list of \albumrow calls. Drop a photograph
% into that directory, re-run assets/make-plates.sh, and the album re-flows
% on its own -- nothing here needs editing.
%
% Both dimensions of every photograph are given because the script has
% already solved the row: each row shares one height, and its widths plus
% gutters span \linewidth exactly. The frame is drawn with \fbox rather than
% tikz so that the 0.3pt rule sits outside the image and the arithmetic the
% script did stays true; it allows for that rule when it computes widths.
\newlength{\albumgutter}\setlength{\albumgutter}{3mm}
\newcommand{\albumgap}{\hspace{\albumgutter}}

\newcommand{\albumphoto}[3]{%
  \begingroup
  \setlength{\fboxsep}{0pt}\setlength{\fboxrule}{0.3pt}%
  \fcolorbox{rulecolor}{white}{\includegraphics[width=#2,height=#3]{#1}}%
  \endgroup
}

% Centred, so that a short final row -- one the script capped rather than
% stretched to the full measure -- sits in the middle of the page instead of
% hanging off the left edge. A full row already spans \linewidth, where
% centring does nothing.
\newcommand{\albumrow}[1]{%
  \par{\centering\leavevmode#1\par}%
  \vspace{\albumgutter}%
}

% The head over the first album page. Quieter than a section title: these
% pages are a continuation of Photographs, not a new section.
\newcommand{\albumhead}[1]{%
  \begin{center}
    {\footnotesize\scshape\color{primary}\so{#1}}\par
    \vspace{2mm}
    {\color{rulecolor}\rule{0.35\linewidth}{0.4pt}}
  \end{center}
  \vspace{3mm}%
}

% A ghosted photograph behind the current page only. The fade is baked into
% the pixels rather than applied as PDF transparency, so the printer's RIP
% receives ordinary opaque artwork -- no flattening, no overprint surprises,
% and what is on screen is what comes off the press.
%
% Two strengths are generated: -lo (7% ink) for the pages that carry dense
% two-column liturgy, and -hi (17%) for the sparse pages that can take more.
% Only the *opening* page of a section is ghosted; interior pages stay clean,
% so the device reads as a section marker rather than as wallpaper.
%
% The same macro places the two full-bleed plates -- the cover and the
% frontispiece that opens the Photographs section -- which are page-sized
% artwork like any other, and are likewise baked rather than layered here.
\newcommand{\pageghost}[1]{%
  \AddToShipoutPictureBG*{%
    \put(0,0){\includegraphics[width=\paperwidth,height=\paperheight]{plates/#1}}%
  }%
}

% --- Tributes --------------------------------------------------------------
% assets/build-tributes.py reads assets/data/tributes.toml -- the words as the
% family sent them -- and writes assets/data/tributes.tex as a run of the
% macros below, which 08-tribute.tex inputs. Edit the TOML and re-run
% assets/make-plates.sh; nothing here needs editing.
%
% A tribute is a letter, so it is set like one: one measure rather than the
% two columns the liturgy uses, no indents, and air between the paragraphs
% instead -- no. These run to four hundred words each, and set across the
% full measure at a size anyone can read they would take a page apiece. So
% the letter itself goes into two columns, like the liturgy, while its title
% and its closing prayer keep the full width: the head stays on one line
% where a 57mm column would have broken it over three, the prayer stays
% centred on the page, and the words in between are set as continuous prose,
% indented the way a book sets a chapter. Air between paragraphs is the one
% thing a letter this long cannot afford -- eight paragraph gaps would cost
% a dozen millimetres of page and buy nothing an indent does not.
\newenvironment{tribute}{%
  \par\begingroup
  \fontsize{9}{10.6}\selectfont
  \setlength{\parskip}{0.5pt}%
  \setlength{\multicolsep}{0pt}%
  \color{primary}%
  %\begin{multicols}{2}
  % \raggedright zeroes \parindent, so the indent is set after it, not before:
  % it is the only thing separating one paragraph from the next here.
  \raggedright\setlength{\parindent}{1.2em}%
  % The letter opens flush, as a first paragraph under a head always does;
  % \ignorespaces then eats the newline that follows \begin{tribute} in the
  % generated file, which would otherwise print as a space before "Dear".
  \noindent\ignorespaces
}{%
  %\end{multicols}%
  \endgroup
}

% soul letterspaces by taking the text apart character by character, and gives
% up on anything that is not a character. The escapes that build-tributes.py
% emits for a name like "Talia & Tyshawn" are exactly that, so they are
% declared here rather than left to break the build on the day someone adds a
% tribute with an ampersand in the title.
\soulregister{\&}{0}
\soulregister{\%}{0}
\soulregister{\$}{0}
\soulregister{\#}{0}
\soulregister{\_}{0}
\soulregister{\textbackslash}{0}
\soulregister{\dots}{0}

% What separates one tribute from the next: a hairline broken by a small
% diamond. It replaces the page break the tributes used to take, and it is
% drawn rather than set from a dingbat so that it uses the booklet's own rule
% colour and hairline weight, and so no font has to be loaded for one glyph.
%
% The title is set on a measure narrower than the page so that a long one --
% "Tribute to My Father-in-Law, Francis Atsekhameh" -- breaks where it reads,
% after the relationship, rather than wherever the full measure happens to
% run out. \@afterheading welds the head to the paragraph beneath it, so a
% column never breaks between a tribute's title and its first line.
\makeatletter
\newcommand{\tributedivider}{%
  \par\addvspace{5mm}%
  {\centering
    \begin{tikzpicture}[inner sep=0pt,outer sep=0pt]
      \draw[rulecolor,line width=0.3pt] (-18mm,0) -- (-2.6mm,0);
      \draw[rulecolor,line width=0.3pt] (2.6mm,0) -- (18mm,0);
      \fill[accent] (0,1mm) -- (1mm,0) -- (0,-1mm) -- (-1mm,0) -- cycle;
    \end{tikzpicture}\par}%
  \nobreak\vspace{3.5mm}%
}

\newcommand{\tributehead}[2]{%

  {\centering
    \begin{minipage}{0.66\linewidth}
      \centering{\footnotesize\scshape\color{primary}\so{#1}}
    \end{minipage}\par
    \vspace{0.8mm}
    {\scriptsize\itshape\color{accent}#2}\par}%
  \nobreak\vspace{1.8mm}%

\makeatother

% The sign-off, set to the right as it is on paper. \\ separates its lines,
% and they are kept tight to each other -- the space the environment puts
% between paragraphs would read as two afterthoughts rather than a signature.
\newcommand{\tributesignoff}[1]{%

  \noindent\begin{minipage}{\linewidth}
    \raggedleft\itshape\color{accent}#1
  \end{minipage}\par
  \addvspace{1.4mm}%
}

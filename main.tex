% Full-bleed cover backdrop. The photograph is scaled to the full paper
% width and top-aligned, so the surplus height falls off the foot of the
% page and is clipped there. Over it: a solid white field at the foot for
% the type, and above that a fade that dissolves the field into the
% photograph so there is no visible edge. Applies to the current page only.
\newlength{\scrimsolid}\setlength{\scrimsolid}{0.41\paperheight}
\newlength{\scrimtop}\setlength{\scrimtop}{0.70\paperheight}

\newcommand{\coverbackdrop}[1]{%
  \AddToShipoutPictureBG*{%
    \put(0,0){%
      \parbox[b][\paperheight][t]{\paperwidth}{%
        \centering\includegraphics[width=\paperwidth]{#1}%
      }%
    }%
    \put(0,0){%
      \begin{tikzpicture}[overlay]
        \fill[white] (0pt,0pt) rectangle (\paperwidth,\scrimsolid);
        \fill[white, path fading=north]
          (0pt,\scrimsolid) rectangle (\paperwidth,\scrimtop);
      \end{tikzpicture}%
    }%
  }%
}
